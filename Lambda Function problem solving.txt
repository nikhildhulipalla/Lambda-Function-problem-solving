is_prime = lambda n: n > 1 and all(n % i != 0 for i in range(2, int(n**0.5) + 1))

# 2
is_armstrong = lambda n: n >= 0 and n == sum(int(d) ** len(str(n)) for d in str(n))

# 3
is_spy = lambda n: n >= 0 and sum(int(d) for d in str(n)) == __import__('math').prod(int(d) for d in str(n))

# 4
is_neon = lambda n: n >= 0 and sum(int(d) for d in str(n * n)) == n

# 5
digit_sum = lambda n: sum(int(d) for d in str(abs(n)))

# 6

# 7

# 8

# 9
is_prime_palindrome = lambda n: is_prime(n) and is_palindrome(n)
